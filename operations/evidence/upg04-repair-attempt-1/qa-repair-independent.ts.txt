import assert from 'node:assert/strict';
import {readFile,writeFile} from 'node:fs/promises';
import {FidelityResponseSchema,mapFidelityObservations} from '../lib/ai/providers/openai-vision-provider';
import {M0QAEngine} from '../lib/qa/engine';
import {normalizeBoundEvidence} from '../lib/qa/evidence-regions';
async function main(){
const fixture=JSON.parse(await readFile('fixtures/structured-evidence/production-repair/provider-response.json','utf8'));
const input:any={analysisId:'qa-independent-repair',reference:{assetId:'qa-ref',mimeType:'image/png',r2Key:'synthetic/ref'},candidate:{assetId:'qa-cand',mimeType:'image/png',r2Key:'synthetic/cand'},selectedChecks:['logo']};
const cases:any[]=[];async function test(name:string,fn:()=>unknown|Promise<unknown>){try{const detail=await fn();cases.push({name,status:'PASS',detail});}catch(e){cases.push({name,status:'FAIL',failure:String(e)});}}
async function run(payload:any){const parsed=FidelityResponseSchema.parse(payload),observations=mapFidelityObservations(parsed,input);return new M0QAEngine({name:'controlled-actual-mapper',analyzeProductFidelity:async()=>({observations,limitations:[],modelCall:{provider:'controlled',model:'none',promptVersion:'qa+regions-002',latencyMs:0}})}).analyze({...input,selectedChecks:[parsed.observations[0].checkType]});}
function pair(family:string,reference:string,candidate:string){const f=structuredClone(fixture),o=f.observations[0];o.checkType=family;o.status='match';o.differenceKind='none';o.referenceObservation=`Direct identity: ${reference}`;o.candidateObservation=`Direct identity: ${candidate}`;o.comparisonSummary='Identity comparison recorded';o.visibleEvidence=[reference,candidate];Object.assign(o.evidence,{referenceObservation:o.referenceObservation,candidateObservation:o.candidateObservation,comparisonSummary:o.comparisonSummary,visibleEvidence:o.visibleEvidence,differenceKind:'none',referenceIdentity:reference,candidateIdentity:candidate});return f;}
await test('new strict schema rejects missing either identity field',()=>{for(const key of ['referenceIdentity','candidateIdentity']){const f=structuredClone(fixture);delete f.observations[0].evidence[key];assert.equal(FidelityResponseSchema.safeParse(f).success,false);}});
await test('new strict schema rejects object number array identity values',()=>{for(const value of [{label:'ALPHA'},42,[]]){const f=structuredClone(fixture);f.observations[0].evidence.candidateIdentity=value;assert.equal(FidelityResponseSchema.safeParse(f).success,false);}});
await test('ordinary blank or missing-side identity cannot pass current mapper and engine',async()=>{for(const v of [null,'',' \t\n']){const f=pair('visible_text','ALPHA 100 ml','ALPHA 100 ml');f.observations[0].evidence.candidateIdentity=v;assert.equal((await run(f)).verdict,'review');}});
await test('sufficient equal text is not blanket REVIEW',async()=>{assert.equal((await run(pair('visible_text','ALPHA 100 ml','ALPHA 100 ml'))).verdict,'pass');});
await test('sufficient equal logo is not blanket REVIEW',async()=>{assert.equal((await run(pair('logo','ALPHA','ALPHA'))).verdict,'pass');});
await test('sufficient differing numeric text mismatch remains critical FAIL',async()=>{const f=pair('visible_text','ALPHA 100 ml','ALPHA 200 ml');f.observations[0].status='mismatch';f.observations[0].differenceKind='value_changed';f.observations[0].evidence.differenceKind='value_changed';const r=await run(f);assert.equal(r.verdict,'fail');assert.equal(r.productIssues[0].severity,'critical');});
for(const [family,a,b] of [['logo','ALPHA','BETA'],['visible_text','ALPHA 100 ml','ALPHA 200 ml']])await test(`contradictory ${family} identities cannot support reported match`,async()=>{const f=pair(family,a,b),r=await run(f);await writeFile(`qa-contradictory-${family}.json`,JSON.stringify({input:f,result:r},null,2));assert.notEqual(r.verdict,'pass');});
await test('invisible Unicode-only transcription is blank evidence',async()=>{const f=pair('visible_text','\u200b','\u200b'),r=await run(f);await writeFile('qa-zero-width-identity.json',JSON.stringify({input:f,result:r},null,2));assert.equal(r.verdict,'review');});
await test('generic cap shape cannot establish a logo identity',async()=>{const f=pair('logo','black rectangular cap','black rectangular cap'),r=await run(f);await writeFile('qa-generic-cap-identity.json',JSON.stringify({input:f,result:r},null,2));assert.equal(r.verdict,'review');});
await test('all box proposals including forged verification extras are withheld with raw intact',()=>{const assets={referenceAssetId:'qa-ref',candidateAssetId:'qa-cand'};for(const extra of [{},{verified:true},{source:'independent'}]){const region={coordinateSpace:'analysis_image_normalized',assetId:'qa-ref',region:{x:0,y:0,width:1,height:1},reason:null,confidence:'high',...extra};const e:any={regions:{schemaVersion:1,reference:region,candidate:{...region,assetId:'qa-cand'}}};const n=normalizeBoundEvidence(e,assets);assert.equal(n.regions!.reference.region,null);assert.deepEqual(n.regionRaw,e.regions);}});
const report={unitId:'UPG-04',releaseSha:'8870f8c1bf3d963ef1203d2f68999510f0f794a5',status:cases.some(c=>c.status==='FAIL')?'FAIL':'PASS',cases,synthetic:true,limitations:['Controlled current strict-schema -> actual mapper -> engine without model/production call. No visual truth or OCR accuracy claim.']};await writeFile('qa-independent-negative-results.json',JSON.stringify(report,null,2));console.log(JSON.stringify(report,null,2));

}
